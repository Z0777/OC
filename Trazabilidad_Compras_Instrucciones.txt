##############################################################################
# 1. Arquitectura y listas de SharePoint
##############################################################################

La app es una sola pantalla, scrTraza, con tres pestañas (Panel, Órdenes de compra y Reglas de días), cinco listas de SharePoint y dos flujos de Power Automate. Toda la lógica de cálculo vive en App.Formulas; App.OnStart solo carga datos e inicializa variables. Sintaxis en configuración regional español: ; separa argumentos, ;; encadena instrucciones, la coma es el decimal y Ungroup usa comillas simples.

  - TRZ_OrdenesCompra
      Qué guarda: Una fila por PO con la fecha de cada paso
      Qué carga la app: Abiertas + pagadas de los últimos 90 días
  - TRZ_HistorialPO
      Qué guarda: Un registro por movimiento o aviso
      Qué carga la app: Solo el de la PO seleccionada
  - TRZ_Config
      Qué guarda: Una fila con los plazos
      Qué carga la app: Completa
  - TRZ_Equipo
      Qué guarda: Persona, correo y área
      Qué carga la app: Activos
  - TRZ_Proveedores
      Qué guarda: Catálogo de proveedores
      Qué carga la app: Activos


Crea las columnas sin espacios ni tildes: el nombre interno es el que se escribe en Power Apps. Etapa, Tipo y Area son texto de una línea, no Elección, para que un valor fuera de la lista nunca se descarte en silencio.

--- Etapas ----------------------------------------------------------------

La etapa se escribe en cada guardado y las fechas límite se calculan en días hábiles (lunes a viernes) en ese momento. Así la app y Power Automate solo comparan fechas.

  - Solicitada
      Entra cuando: Se registra la PO
      Siguiente paso: Magda confirma la recepción
      Alerta roja: —
  - Esperando factura
      Entra cuando: Bodega confirma y no hay factura
      Siguiente paso: Proveedor; luego Zoila llama
      Alerta roja: Hoy ≥ FechaAlertaFactura
  - Factura sin recepción
      Entra cuando: Entra la factura antes que el producto
      Siguiente paso: Magda confirma si llegó
      Alerta roja: Desde el día siguiente
  - Pendiente 101
      Entra cuando: Hay recepción y factura
      Siguiente paso: Lorena da el 101
      Alerta roja: Hoy > FechaLimite101
  - En crédito
      Entra cuando: Hay 101 y factura
      Siguiente paso: Cuentas por pagar
      Alerta roja: Hoy ≥ FechaPagoEstimada (aviso amarillo)
  - Pagada
      Entra cuando: Se registra el pago
      Siguiente paso: —
      Alerta roja: —


&#91;embedded content: Recorrido de una PO · 5 etapas, 1 desvío y el regreso por rechazo]

La ruta normal baja por la columna; si la factura entra antes que el producto, la PO pasa por Factura sin recepción hasta que Magda confirma. Rechazar la factura la regresa a Esperando factura.

##############################################################################
# 3. App.Formulas y App.OnStart
##############################################################################

App.Formulas calcula todo lo que se ve (etapa, alerta, responsable, días, filtros, indicadores) y se recalcula solo cuando cambia una colección o variable. App.OnStart carga cinco listas en paralelo e inicializa las variables. Ningún control guarda lógica de cálculo propia.

Antes de pegar, en Configuración:

- General › Límite de filas de datos: 2000.
- Pantalla: Escalar para ajustar desactivado; Bloquear relación de aspecto desactivado.
- Actualizaciones: Fórmulas con nombre activado.
- Datos: agrega TRZ_OrdenesCompra, TRZ_HistorialPO, TRZ_Config, TRZ_Equipo y TRZ_Proveedores.
- App.StartScreen: scrTraza.

// 4. ÓRDENES CALCULADAS (una fila por PO cargada)

frOrdenes =
    ForAll(colOrdenes As o;
        With(
            {
                fRef: Switch(o.Etapa;
                    "Solicitada"; o.FechaSolicitud;
                    "Esperando factura"; o.FechaRecepcion;
                    "Factura sin recepción"; o.FechaFactura;
                    "Pendiente 101"; If(o.FechaFactura > o.FechaRecepcion; o.FechaFactura; o.FechaRecepcion);
                    "En crédito"; o.Fecha101;
                    o.FechaPago);
                alertaFac: o.Etapa = "Esperando factura" And !IsBlank(o.FechaAlertaFactura) And varHoy >= o.FechaAlertaFactura;
                atraso101: o.Etapa = "Pendiente 101" And !IsBlank(o.FechaLimite101) And varHoy > o.FechaLimite101;
                pagoVencido: o.Etapa = "En crédito" And !IsBlank(o.FechaPagoEstimada) And varHoy >= o.FechaPagoEstimada;
                llam: Coalesce(o.Llamadas; 0)
            };
            With(
                {
                    dias: If(IsBlank(fRef) Or fRef >= varHoy; 0;
                        CountRows(Filter(ForAll(Sequence(DateDiff(fRef; varHoy; TimeUnit.Days)) As s; {F: DateAdd(fRef; s.Value; TimeUnit.Days)}); Weekday(F; StartOfWeek.Monday) <= 5)));
                    nivel: Switch(o.Etapa;
                        "Pagada"; "ok";
                        "En crédito"; If(pagoVencido; "warn"; "info");
                        "Factura sin recepción"; If(varHoy > o.FechaFactura; "crit"; "warn");
                        "Pendiente 101"; If(atraso101; "crit"; "warn");
                        "Esperando factura"; If(alertaFac; "crit"; "info");
                        "neutral");
                    resp: Switch(o.Etapa;
                        "Solicitada"; "Bodega";
                        "Factura sin recepción"; "Bodega";
                        "Pendiente 101"; "Compras";
                        "En crédito"; "Pagos";
                        "Esperando factura"; If(alertaFac Or llam > 0; "Proveedores"; "Proveedor");
                        "")
                };
                {
                    ID: o.ID;
                    PO: o.Title;
                    Proveedor: o.Proveedor;
                    Descripcion: o.Descripcion;
                    Cantidad: o.Cantidad;
                    NumFactura: o.NumFactura;
                    FechaSolicitud: o.FechaSolicitud;
                    UltimoMovimiento: o.UltimoMovimiento;
                    Etapa: o.Etapa;
                    Nivel: nivel;
                    NivelOrden: Switch(nivel; "crit"; 0; "warn"; 1; "info"; 2; "neutral"; 3; 4);
                    Dias: dias;
                    Resp: resp;
                    RespNombre: Switch(resp;
                        "Bodega"; frPersonaBodega & " (Bodega)";
                        "Compras"; frPersonaCompras & " (Compras)";
                        "Proveedores"; frPersonaProveedores & " (Atención a proveedores)";
                        "Pagos"; frPersonaPagos;
                        "Proveedor"; "Proveedor";
                        "");
                    Accion: Switch(o.Etapa;
                        "Solicitada"; "Confirmar ingreso del producto en bodega";
                        "Factura sin recepción"; "Confirmar si el producto llegó a bodega";
                        "Pendiente 101"; If(atraso101; "101 atrasado: " & dias & " días hábiles"; "Dar el 101 en el sistema");
                        "En crédito"; If(pagoVencido; "Pago vencido desde el "; "Pago estimado el ") & Text(o.FechaPagoEstimada; "dd/mm/yyyy");
                        "Esperando factura";

                    Llamadas: llam;
                    Rechazos: Coalesce(o.Rechazos; 0);
                    HRec: !IsBlank(o.FechaRecepcion);
                    HFac: !IsBlank(o.FechaFactura);
                    H101: !IsBlank(o.Fecha101);
                    HPag: !IsBlank(o.FechaPago);
                    DiasRecFac: If(IsBlank(o.FechaRecepcion) Or IsBlank(o.FechaFactura) Or o.FechaFactura <= o.FechaRecepcion; 0;

##############################################################################
# 0. Cómo leer y árbol completo
##############################################################################

Cada tabla es un contenedor o un grupo de controles hermanos. Columna Control = nombre exacto; Propiedad = nombre tal como aparece en la lista de propiedades de Power Apps; Valor = lo que se pega en la barra de fórmulas, empezando con =.

Reglas de construcción:

- Usa Contenedor vertical u horizontal (no el contenedor libre) para todo lo que diga "vertical" u "horizontal".
- Dentro de un contenedor con diseño, X e Y no se usan. El tamaño lo deciden FillPortions (0 = tamaño fijo; 1 o más = flexible), Width, Height, LayoutMinWidth y AlignInContainer.
- Dentro de una plantilla de galería sí se usan X, Y, Width y Height.
- Orden en la vista de árbol: dentro de un contenedor con diseño, de arriba hacia abajo = de primero a último en pantalla. En la pantalla y en las plantillas de galería, el de arriba queda al frente.
- Fuente de todos los textos: =Font.'Segoe UI', salvo donde diga Font.'Courier New'.
- Para crear rápido controles con el mismo estilo, configura el primero, cópialo (Ctrl+C, Ctrl+V) y cámbiale el nombre y las propiedades que difieren.

Cambios respecto a la sección 4 de la Guía general: conDetContenido se elimina (sus secciones van directo en conDetalle para que el desplazamiento funcione) y se agregan conSigCaja, conFormCaja, conPlazosHead, lblPlazosSub, lblEtFondo, rctColaDiv, rctIndDiv, rctEqDiv, lblProvNota, lblHistVacio y los textos de cada fila de plazos.

--- Por qué al hacer clic solo aparece tu nombre --------------------------

La lista de drpVista sale de frOpcionesVista, que arma: primero tu usuario y después cada persona de la colección colEquipo (lista TRZ_Equipo con Activo = Sí). Si solo aparece tu nombre, colEquipo no tiene a nadie más. Revisa en este orden:

1. TRZ_Equipo debe tener una fila por persona: Lorena (Area = Compras), Magda (Area = Bodega), Zoila (Area = Proveedores), la persona de pagos (Area = Pagos) y tú (Area = Admin). El correo en Email debe ser el correo con el que cada quien entra a Microsoft 365.
2. La columna Activo de cada fila debe estar en Sí. Las filas creadas antes de agregar la columna quedan vacías y el filtro Activo = true las descarta. En la configuración de la columna Activo pon "Valor predeterminado: Sí" y marca Sí en las filas existentes.
3. La columna Area debe decir exactamente Compras, Bodega, Proveedores, Pagos o Admin (sin espacios al final).
4. colEquipo se carga al abrir la app. Después de agregar o corregir personas, vuelve a cargar: en Power Apps Studio, menú "…" de App › Ejecutar OnStart; con la app publicada, toca el botón de actualizar (icoRefrescar) o cierra y abre la app.
5. Diagnóstico rápido: agrega temporalmente una etiqueta lblDiagEquipo con Text =CountRows(colEquipo) & " personas en colEquipo · " & CountRows(frOpcionesVista) & " opciones en la lista". Si dice 0 o 1 personas, el problema está en la lista TRZ_Equipo (pasos 1 a 3). Bórrala al terminar.
6. En el cuadro combinado moderno, el texto que escribes filtra la lista. Para ver todas las opciones, haz clic en la flecha de la derecha o borra el texto del cuadro antes de abrirlo.


--- Cómo se usa "Trabajando como" -----------------------------------------

1. Al abrir la app, drpVista muestra tu nombre con "(tú)", porque varVistaEmail empieza con tu correo.
2. Haz clic en la flecha de drpVista: aparece tu nombre primero y debajo cada persona del equipo como "Nombre · Área" (por ejemplo "Magda · Bodega").
3. Elige una persona: OnChange cambia varVistaEmail, varVistaNombre y varVistaArea. El Panel muestra los pendientes de esa persona y en la ficha de la PO se activan las acciones de su área.
4. Si tu área es Admin, puedes registrar acciones en la vista de otra persona; el historial guarda tu nombre real y "vista de Magda". Si no eres Admin, la vista de otra persona es solo de consulta (lblModo dice "Solo lectura").
5. Para volver, elige de nuevo tu nombre en drpVista.

Set(varGuardando; true);;

        varModoPO = "nueva" And !IsBlank(LookUp(TRZ_OrdenesCompra; Title = po)); Set(varErrPO; "La PO " & po & " ya está registrada. Búscala en Órdenes de compra.");

        // ---------- NUEVA PO ----------
        varModoPO = "nueva";
            Set(varErrPO; "");;
            Set(varRes; Patch(TRZ_OrdenesCompra; Defaults(TRZ_OrdenesCompra);
                {
                    Title: po;
                    Proveedor: prov;

                    Llamadas: 0;
                    Rechazos: 0;
                    Etapa: "Solicitada";
                    Cerrada: 0;
                    UltimoMovimiento: Now()
                }
            ));;
            If(
                IsEmpty(Errors(TRZ_OrdenesCompra));
                    Collect(colOrdenes; varRes);;
                    Patch(TRZ_HistorialPO; Defaults(TRZ_HistorialPO); frFirma;
                        {
                            Title: po; OrdenID: varRes.ID; Tipo: "Solicitud"; FechaEvento: varRes.FechaSolicitud;
                            Detalle: "PO registrada con estatus Solicitado" &
                                If(IsBlank(varRes.Cantidad); ""; " · " & varRes.Cantidad) &
                                If(IsBlank(varRes.PdfUrl); ""; " · PDF adjunto")
                        }
                    );;
                    If(IsBlank(LookUp(colProveedores; Lower(Title) = Lower(prov)));
                        Collect(colProveedores; Patch(TRZ_Proveedores; Defaults(TRZ_Proveedores); {Title: prov; Activo: true}))
                    );;
                    Set(varPOId; varRes.ID);;
                    ClearCollect(colHist; Sort(Filter(TRZ_HistorialPO; OrdenID = varPOId); ID; SortOrder.Descending));;

        Set(varPOAct; LookUp(TRZ_OrdenesCompra; ID = varPOId));;
        Set(varCambios;
            Concat(
                Filter(
                    Table(
                        {C: "Proveedor"; A: varPOAct.Proveedor; B: prov};
                        {C: "Descripción"; A: varPOAct.Descripcion; B: Trim(txtDescripcion.Text)};
                        {C: "Cantidad"; A: varPOAct.Cantidad; B: Trim(txtCantidad.Text)};

        );;

            ));;

                IsEmpty(Errors(TRZ_OrdenesCompra));

                    );;

                    ClearCollect(colHist; Sort(Filter(TRZ_HistorialPO; OrdenID = varPOId); ID; SortOrder.Descending));;
                    Set(varErrPO; "");;
                    Set(varModalPO; false);;
                    Notify("Datos de la PO actualizados."; NotificationType.Success);
                Notify("No se guardó: " & First(Errors(TRZ_OrdenesCompra)).Message; NotificationType.Error)
            )
        )
    )
);;
